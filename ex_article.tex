\section{Introduction}
The incompressible Navier-Stokes equations (NSE) are a fundamental model of incompressible fluid flow and provide a canonical testbed for numerical methods designed for long-time simulations. 
Let $\Omega\subset\mathbb R^2$ be a bounded, connected domain with a $C^2$ boundary. 
In this paper, we consider the forced incompressible NSE in dimensionless form, subject to the no-slip boundary conditions:
\begin{equation}\label{eq:nse}
\begin{aligned}
 u_t-\nu\Delta u+(u\cdot\nabla)u+\nabla p&=f
 &&\text{in }\Omega\times(0,\infty),\\
 \nabla\cdot u&=0
 &&\text{in }\Omega\times(0,\infty),\\
 u&=0 &&\text{on }\partial\Omega\times(0,\infty),\\
 u(\cdot,0)&=u_0 &&\text{in }\Omega.
\end{aligned}
\end{equation}
Here, $u$ and $p$ denote the velocity and pressure, respectively, $\nu>0$ is the dimensionless viscosity parameter, $f$ is the external body force, and $u_0$ is the given initial velocity. The pressure is normalized by requiring $\int_\Omega p(x,t)\,dx=0$ for each $t>0$.

\section{Preliminaries and the mean reverting ccSAV reformulation of the NSE}
In this section, we first introduce the functional setting and notation for the incompressible Navier-Stokes equations. Then we introduce the mean-reverting concurrently corrected scalar auxiliary variable (mr-ccSAV) reformulation of the NSE, which is the basis for the numerical time stepping scheme and the corresponding stability and convergence analysis in the subsequent sections. 

\subsection{Function spaces and notation}\label{sec:functional-setting}
For the mathematical setting of the problem, we consider the following space
\begin{equation}
  \begin{aligned}
 &V=\{v\in H_0^1(\Omega)^2: \nabla\cdot v=0\}, \\
 &H=\{v \in L^2(\Omega)^2: \nabla \cdot v=0, v\cdot n = 0 \text{ on } \partial\Omega\}
  \end{aligned}
\end{equation}
We denote the $L^2(\Omega)^2$ inner product and norm by
$(\cdot,\cdot)$ and $|\cdot|$, respectively, and equip $V$ with
the norm $\|v\|=|\nabla v|$. Let $V'$ be the dual of $V$, with
duality pairing $\langle\cdot,\cdot\rangle$ and norm
$\|g\|_{V'}=\sup_{0\ne v\in V}|\langle g,v\rangle|/\|v\|$.
Identifying $H$ with its dual through the $L^2$ inner product,
we obtain the dense and continuous embeddings
\[
V\hookrightarrow H\equiv H'\hookrightarrow V',
\]
where $\langle g,v\rangle=(g,v)$ for $g\in H$ and $v\in V$.
For the pressure, we use the mean-zero space
\[
L_0^2(\Omega)
=\left\{q\in L^2(\Omega):
          \int_\Omega q\,dx=0\right\}.
\]

Let $\mathcal{P}:L^2(\Omega)^2\to H$ be the orthogonal Leray projection.
The Stokes operator is defined by
\[
 A=-\mathcal{P}\Delta,\qquad D(A)=H^2(\Omega)^2\cap V,
\]
which is a positive self-adjoint operator on $H$ with compact inverse.
Its extension from $V$ to $V'$ is defined by
\[
\langle Au,v\rangle=(\nabla u,\nabla v),
\qquad \text{for all }u,v\in V.
\]

Writing $\lambda_1>0$ for the first eigenvalue of $A$, the Poincaré inequality and Stokes regularity give the estimates
\begin{equation}\label{eq:stokes-estimates}
\begin{aligned}
\lambda_1|v|^2 &\le \|v\|^2,
&&\text{for all }v\in V,\\
\|v\|_{H^2} &\le C|Av|,
&&\text{for all }v\in D(A),
\end{aligned}
\end{equation}
where $C$ depends only on $\Omega$. 

For $u,v,w\in V$, define the trilinear form
\[
 b(u,v,w)=\int_\Omega (u\cdot\nabla)v\cdot w\,dx,
 \qquad \langle B(u,v),w\rangle=b(u,v,w).
\]
The divergence-free constraint and homogeneous boundary conditions imply
\begin{equation}\label{eq:skew-symmetry}
 b(u,v,w)=-b(u,w,v),\qquad b(u,v,v)=0.
\end{equation}
The two-dimensional interpolation inequalities and Stokes regularity
also give
\begin{equation}\label{eq:trilinear-estimates}
  |b(u,v,w)| \le \left\{
      \begin{aligned}
            &C|u|^{1/2}\|u\|^{1/2}\|v\||w|^{1/2}\|w\|^{1/2},&&u,v,w\in V,\\
            &C|u|^{1/2}|Au|^{1/2}\|v\||w|,&&u\in D(A),\ v\in V,\ w\in H.
      \end{aligned}
  \right.    
\end{equation}
The above estimates define a continuous bilinear operator
$B:V\times V\to V'$. If $(u\cdot\nabla)v\in L^2(\Omega)^2$,
this definition agrees with $B(u,v)=\mathcal P((u\cdot\nabla)v)\in H$.

Throughout this section, assume
\begin{equation}\label{eq:continuous-data}
 u_0\in V,\qquad f\in L^\infty(0,\infty;L^2(\Omega)^2),\qquad
 F=\mathcal{P}f.
\end{equation}
Applying $\mathcal{P}$ to \cref{eq:nse} yields the projected problem
\begin{equation}\label{eq:projected-nse}
 u_t+\nu Au+B(u,u)=F,\qquad u(0)=u_0.
\end{equation}
This formulation provides the basis for the numerical scheme and its analysis developed in the subsequent sections.

Under \cref{eq:continuous-data}, the two-dimensional problem admits
a unique global strong solution satisfying
\begin{equation}\label{eq:nse-energy}
\frac12\frac{d}{dt}|u|^2+\nu\|u\|^2=(F,u)
\end{equation}
for almost every $t>0$.
Moreover, the solution is uniformly bounded in $H$ and $V$:
\begin{equation}\label{eq:nse-uniform-bounds}
\sup_{t\ge0}|u(t)|^2\le C_0,\qquad
\sup_{t\ge0}\|u(t)\|^2\le C_1,
\end{equation}
where $C_0=C_0(\nu,\Omega,F_*,|u_0|)$,
$C_1=C_1(\nu,\Omega,F_*,\|u_0\|)$, and $F_*=\operatorname*{ess\,sup}_{t\ge0}|F(t)|$.
For sufficiently large time, these bounds become independent
of the initial data. More precisely, there exist radii
$R_0,R_1>0$, depending only on $\nu$, $\Omega$, and $F_*$,
and an entry time $T_*=T_*(\nu,\Omega,F_*,|u_0|)$ such that
\begin{equation}\label{eq:nse-absorbing}
|u(t)|\le R_0,\qquad
\|u(t)\|\le R_1,\qquad t\ge T_*.
\end{equation}
For background on these estimates and absorbing properties,
see \cite[Chapter~III, Section~2.2]{Temam1997}.

These long-time bounds motivate the numerical scheme
developed in the subsequent sections. Under the stated data assumptions, we establish
discrete $L^2$ absorption on arbitrary positive time grids with
$t_n\to\infty$. If the step sizes and their successive growth ratios
are uniformly bounded, we further obtain uniform-in-time $H^1$
bounds and $H^1$ absorbing bounds independent of the initial data.
No additional small-step stability restriction is required.

\subsection{Mean-reverting ccSAV reformulation}
\label{sec:mr-ccsav}

Starting from \cref{eq:projected-nse}, we introduce an auxiliary
scalar $r(t)$ and consider the extended system
\begin{equation}\label{eq:mr-ccsav}
\begin{aligned}
&u_t+\nu Au+G_u(r)B(u,u)=F,\\
&r_t+\gamma r+G_r(r)\langle B(u,u),u\rangle=0,
\end{aligned}
\end{equation}
with initial conditions $u(0)=u_0$ and $r(0)=0$.
Here $\gamma>0$ is the mean-reversion parameter.
For an integer $k\ge1$, define
\begin{equation}\label{eq:ccsav-factors}
G_u(r)=1-r^k,\qquad
G_r(r)=\sum_{j=0}^{k-1}r^j,
\end{equation}
so that
\[
G_u(r)=(1-r)G_r(r).
\]
The integer $k$ is chosen to match the temporal order of the underlying
discretization.

By \cref{eq:skew-symmetry}, the no-slip convective term satisfies
\[
\langle B(u,u),u\rangle=0,\qquad u\in V.
\]
Hence the scalar equation reduces to $r_t+\gamma r=0$, and
$r(t)=r(0)e^{-\gamma t}$.
In particular, $r\equiv0$ for the prescribed initialization,
and \cref{eq:mr-ccsav} reduces to \cref{eq:projected-nse}.

Since the correction factors satisfy
$G_u(r)+(r-1)G_r(r)=0$, pairing the velocity equation with $u$
and multiplying the scalar equation by $r-1$ cancels the
convective contributions. This cancellation underlies the
discrete energy estimate. 
At the discrete level, the mean-reverting term $\gamma r$
damps the auxiliary variable toward the equilibrium $r=0$
over long times. Moreover, if the numerical auxiliary
variable satisfies $r=O(\tau)$, then
$G_u(r)-1=-r^k=O(\tau^k)$, motivating the choice of correction
factors for an order-$k$ discretization.

The combination of mean reverting and concurrent correction,
abbreviated as mr-ccSAV, was first introduced for variable-step ETD
schemes in \cite{wang2026unconditionally}.
It builds on the mean-reverting SAV formulations of
\cite{han2025highly,coleman2024efficient},
the concurrent-correction idea of Hou and Qiao
\cite{hou2023implicit}, and the zero-energy-preservation
(ZEC) approach
\cite{li2022new,yang2021novel,yang2021numerical};
see also \cite{huang2022new} for related higher-order
sequential correction strategies. 
For comparison, setting $\gamma=0$ and $k=1$ gives
a SAV-ZEC-type formulation used in several fluid models
\cite{yang2021new,yang2021novel,zhang2024unified,li2022new}.

\section{Algorithm construction and stage solvability}
\label{sec:alg}

Based on the mr-ccSAV reformulation \eqref{eq:mr-ccsav},
we construct a family of singly diagonally implicit Runge--Kutta
(SDIRK) schemes that treat viscosity implicitly and convection
explicitly. The same Stokes operator is reused across all stages
within each time step. We present the SDIRK-mr-ccSAV schemes
below and then establish their stagewise solvability.

\subsection{The SDIRK-mr-ccSAV scheme}
\label{sec:sdirk-scheme}
We first introduce the standard and incremental forms of
an IMEX RK method. Let $0=t_0<t_1<\cdots$ be a arbitrary time grid,
with $\tau_n=t_{n+1}-t_n>0$. For an evolution equation of the form
\[
y_t=\mathcal L y+\mathcal N(t,y),
\]
set $Y_0=y^n$ and denote the $s$ unknown stages by
$Y_1,\ldots,Y_s$.  The standard IMEX Runge--Kutta stage equations are
\begin{equation}\label{eq:standard-imex-stages}
\begin{aligned}
Y_i =Y_0+\tau_n\sum_{j=1}^{i}
  a_{ij}^{\mathrm{RK}}\mathcal L Y_j +\tau_n\sum_{j=0}^{i-1}
  \widehat a_{ij}^{\mathrm{RK}}
  \mathcal N(t_n+c_j\tau_n,Y_j),
\qquad i=1,\ldots,s.
\end{aligned}
\end{equation}
The implicit and explicit coefficient matrices are given, repectively, by
$\mathsf A^{\mathrm{RK}}$ and
$\widehat{\mathsf A}^{\mathrm{RK}}$, respectively, as
\begin{equation*}\label{eq:imex-rk-matrices}
\mathsf A^{\mathrm{RK}}
=
\begin{pmatrix}
a_{11}^{\mathrm{RK}} & 0 & \cdots & 0\\
a_{21}^{\mathrm{RK}} & a_{22}^{\mathrm{RK}} & \ddots & \vdots\\
\vdots & \ddots & \ddots & 0\\
a_{s1}^{\mathrm{RK}} & \cdots
& a_{s,s-1}^{\mathrm{RK}} & a_{s,s}^{\mathrm{RK}}
\end{pmatrix},
\ \widehat{\mathsf A}^{\mathrm{RK}}
=
\begin{pmatrix}
\widehat a_{10}^{\mathrm{RK}} & 0 & \cdots & 0\\
\widehat a_{20}^{\mathrm{RK}}
& \widehat a_{21}^{\mathrm{RK}} & \ddots & \vdots\\
\vdots & \ddots & \ddots & 0\\
\widehat a_{s0}^{\mathrm{RK}} & \cdots
& \widehat a_{s,s-2}^{\mathrm{RK}}
& \widehat a_{s,s-1}^{\mathrm{RK}}
\end{pmatrix},
\end{equation*}
and the stage abscissae $c_i = \sum_{j=1}^i a_{ij}^{\mathrm{RK}} = \sum_{j=0}^{i-1} \widehat a_{ij}^{\mathrm{RK}}$, with $c_0=0$.

Inspired by the work of Liao et al.~\cite{liao2026longtime},
we rewrite the IMEX RK method in an equivalent incremental
form to facilitate both stability analysis and numerical
implementation. To derive this form, we retain the first
stage equation and, for $i=2,\ldots,s$, subtract the equation
for stage $i-1$ from that for stage $i$. Define the lower triangular matrix
\[
E=(e_{ij})_{i,j=1}^s,
\qquad
e_{ij}=
\begin{cases}
1,&j\le i,\\
0,&j>i.
\end{cases}
\]
Then the incremental coefficient matrices are
\begin{equation}\label{eq:incremental-rk-matrices}
\mathsf A=(a_{ij})=E^{-1}\mathsf A^{\mathrm{RK}},
\qquad
\widehat{\mathsf A}=(\widehat a_{ij})
=E^{-1}\widehat{\mathsf A}^{\mathrm{RK}},
\end{equation}
and the stage equations become
\begin{equation}\label{eq:incremental-imex-stages}
\begin{aligned}
Y_i-Y_{i-1}
=\tau_n\sum_{j=1}^{i}a_{ij}\mathcal L Y_j +\tau_n\sum_{j=0}^{i-1}
  \widehat a_{ij}\mathcal N(t_n+c_j\tau_n,Y_j),
\qquad i=1,\ldots,s.
\end{aligned}
\end{equation}
The standard and incremental forms are algebraically equivalent.
We use this incremental form as the starting point for constructing and analyzing the numerical schemes in the rest of this paper.

On each interval $[t_n,t_{n+1}]$, let $u_{n,i}$ and $r_{n,i}$ denote the stage approximations to $u(t_{n}+c_i\tau_n)$ and $r(t_{n}+c_i\tau_n)$, respectively, for $i=0,\ldots,s$. 
We write $u^n$ and $r^n$ for the numerical solution at time $t_n$. 
For convenience, define
$$
F_{n,j} := F(t_n+c_j\tau_n),\quad B_{n,j} = B(u_{n,j},u_{n,j}), \qquad j=0,\ldots,s-1.
$$

Starting from $(u_{n,0}, r_{n,0}) = (u^n, r^n)$, we compute the stage values $(u_{n,i}, r_{n,i})$ for $i=1,\ldots,s$ by solving the following system of equations:
\begin{subequations}\label{eq:sdirk-mr-ccsav}
\begin{align}
&u_{n,i}-u_{n,i-1}
+\nu\tau_n\sum_{j=1}^{i}a_{ij}Au_{n,j}
=\tau_n\sum_{j=0}^{i-1}\widehat a_{ij}\bigl( F_{n,j}
  -G_u(r_{n,i}) B_{n,j}\bigr),\\
&r_{n,i}-r_{n,i-1}
+\gamma\tau_n\sum_{j=1}^{i}a_{ij}r_{n,j}
=-\tau_n \sum_{j=0}^{i-1} \widehat{a}_{ij} G_r(r_{n,i})
  \langle B_{n,j},u_{n,i}\rangle,
\end{align}
\end{subequations}
and the numerical solution at $t_{n+1}$ is defined by $(u^{n+1}, r^{n+1}) = (u_{n,s}, r_{n,s})$.

\begin{remark}
This work focuses on the singly diagonally implicit coefficient pairs of orders one, two, and three which have the following coefficient matrices:
\begin{enumerate}
\item One-stage, first-order:
\[
\mathsf A^{\mathrm{RK}}=(1),
\qquad
\widehat{\mathsf A}^{\mathrm{RK}}=(1).
\]

\item Two-stage, second-order:
\[
\mathsf A^{\mathrm{RK}}
=
\begin{pmatrix}
1-\frac{1}{\sqrt{2}} & 0\\
\frac{1}{\sqrt{2}} & 1-\frac{1}{\sqrt{2}}
\end{pmatrix},
\qquad
\widehat{\mathsf A}^{\mathrm{RK}}
=
\begin{pmatrix}
1-\frac{1}{\sqrt{2}} & 0\\
-\frac{1}{\sqrt{2}} & 1+\frac{1}{\sqrt{2}}
\end{pmatrix}.
\]

\item Four-stage, third-order:
\[
\mathsf A^{\mathrm{RK}}
=
\begin{pmatrix}
\frac14 & 0 & 0 & 0\\
\frac{11}{20} & \frac14 & 0 & 0\\
\frac9{22} & \frac{15}{44} & \frac14 & 0\\
\frac{56}{99} & \frac{25}{66} & -\frac7{36} & \frac14
\end{pmatrix},
\qquad
\widehat{\mathsf A}^{\mathrm{RK}}
=
\begin{pmatrix}
\frac14 & 0 & 0 & 0\\
-\frac{41}{100} & \frac{121}{100} & 0 & 0\\
\frac3{16} & \frac12 & \frac5{16} & 0\\
\frac16 & \frac8{33} & \frac{25}{33} & -\frac16
\end{pmatrix}.
\]
\end{enumerate}

For these three coefficient pairs, we take $k=1,2,3$,
respectively, in \eqref{eq:ccsav-factors}, matching the
temporal order of the underlying IMEX RK method.
For each pair, all implicit diagonal entries are equal.
We refer to the resulting methods as {\bf SDIRK-mr-ccSAV}.
Since the incremental reformulation preserves the diagonal
coefficients, every stage within a time step uses the same
Stokes operator, allowing its factorization to be reused throughout the step.
\end{remark}

\subsection{Stagewise solvability}
In this section, we first decouple each stage of \eqref{eq:sdirk-mr-ccsav} into two linear Stokes solves with a common operator and finding a real root of a scalar polynomial. Using this formulation, we establish the existence of stage solutions for arbitrary positive time steps and their uniqueness for sufficiently small time steps.

For a stage $i \in \{1, \ldots, s\}$ and suppose that the previous stage values $(u_{n,j}, r_{n,j})$ for $j=0, \ldots, i-1$ are known. Define 
$$
\widetilde{F}_{n,i} := \sum_{j = 0}^{i-1} \widehat{a}_{ij} F_{n,j},\qquad
\widetilde{B}_{n,i} := \sum_{j = 0}^{i-1} \widehat{a}_{ij} B_{n,j}.
$$
Then velocity equation at stage $i$ \eqref{eq:sdirk-mr-ccsav} can be rewritten as
\begin{equation}\label{eq:stage-velocity}
\left(I + \nu \tau_n a_{ii} A\right) u_{n,i} = R_{n,i}^{u} - \tau_n G_u(r_{n,i}) \widetilde{B}_{n,i}.
\end{equation}
where 
$$
R_{n,i}^{u} :=  u_{n,i-1} - \nu \tau_n \sum_{j=1}^{i-1} a_{ij}A u_{n,j} + \tau_n \widetilde{F}_{n,i}
$$
By \eqref{eq:stage-velocity}, for any given scalar trial value of $r$, the velocity value $u_{n,i}$ can be represented in the form 
$$
u_{n,i} = u_{n,i}^{(1)} - \tau_n G_u(r) u_{n,i}^{(2)}, 
$$
where $u_{n,i}^{(1)}$ and $u_{n,i}^{(2)}$ can be determined by the following two linear Stokes problems:
\begin{equation}\label{eq:stage-linear-stokes}
      (I + \nu \tau_n a_{ii} A) u_{n,i}^{(1)} = R_{n,i}^{u},\qquad
      (I + \nu \tau_n a_{ii} A) u_{n,i}^{(2)} = \widetilde{B}_{n,i}.
\end{equation}

\begin{remark}
For implementation, the two shifted Stokes problems in
\eqref{eq:stage-linear-stokes} can be written in
velocity--pressure form with no-slip boundary conditions as
\begin{equation}\label{eq:stage-stokes-vp}
u_{n,i}^{(\ell)}-\nu\tau_n a_{ii}\Delta u_{n,i}^{(\ell)}
  +\nabla p_{n,i}^{(\ell)}
=g_{n,i}^{(\ell)},\qquad
\nabla\cdot u_{n,i}^{(\ell)}=0, \qquad \ell=1,2,
\end{equation}
where the known right-hand sides are
\begin{equation*}
g_{n,i}^{(1)} =u_{n,i-1}
  +\nu\tau_n\sum_{j=1}^{i-1}a_{ij}\Delta u_{n,j}
  +\tau_n\sum_{j=0}^{i-1}\widehat a_{ij}f_{n,j},\qquad
g_{n,i}^{(2)} =\sum_{j=0}^{i-1}\widehat a_{ij}
  (u_{n,j}\cdot\nabla)u_{n,j}.
\end{equation*}
Here $f_{n,j}$ denotes the unprojected stage force satisfying
$\mathcal P f_{n,j}=F_{n,j}$. The pressure multipliers are normalized
by $\int_\Omega p_{n,i}^{(\ell)}\,dx=0$.
After spatial discretization, standard solvers for shifted
Stokes systems can be used.
Since all implicit diagonal entries are equal, the same discrete operator is used
for both solves at every stage, allowing its factorization
or preconditioner to be reused throughout each time step.
\end{remark}

To determine the stage value $r_{n,i}$, we substitute the above representation of $u_{n,i}$ into the scalar equation in \eqref{eq:sdirk-mr-ccsav} and obtain a polynomial equation for $r_{n,i}$:
\begin{equation}\label{eq:stage-scalar}
r_{n,i} - r_{n,i-1} + \gamma \tau_n \sum_{j=1}^{i} a_{ij} r_{n,j} + \tau_n G_r(r_{n,i}) \langle \widetilde{B}_{n,i}, u_{n,i}^{(1)} - \tau_n G_u(r_{n,i}) u_{n,i}^{(2)} \rangle = 0.
\end{equation}
Define the known scalar quantities
$$
R_{n,i}^{r} := r_{n,i-1} - \gamma \tau_n \sum_{j=1}^{i-1} a_{ij} r_{n,j},\quad \alpha_{n,i}:= \langle \widetilde{B}_{n,i}, u_{n,i}^{(1)} \rangle,\quad \beta_{n,i}:= \tau_n \langle \widetilde{B}_{n,i}, u_{n,i}^{(2)} \rangle.
$$
Then $r_{n,i}$ is determined by choosing a real root of the polynomial equation
\begin{equation}\label{eq:stage-scalar-polynomial}
      \mathcal{F}_{n,i}(r) := (1 + \gamma \tau_n a_{i,i})r + \tau_n G_r(r) (\alpha_{n,i} - G_u(r) \beta_{n,i}) - R_{n,i}^{r} = 0.
\end{equation}
which has degree at most $2k-1$. We should state here that the existence and the stability analysis below hold for any real root of \eqref{eq:stage-scalar-polynomial}.
